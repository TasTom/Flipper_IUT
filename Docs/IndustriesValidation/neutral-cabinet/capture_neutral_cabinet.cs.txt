if(!UnityEditor.EditorApplication.isPlaying)throw new Exception("Play attendu");
var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();
if(scene.name!="Industries")throw new Exception("Industries attendue");
var display=UnityEngine.Object.FindAnyObjectByType<BackglassDisplay>();
var panel=(RectTransform)display.transform;var model=panel.Find("Pinball_Cabinet");
if(model==null)throw new Exception("Cabinet absent");
var rules=UnityEngine.Object.FindAnyObjectByType<IndustriesVpeGame>();
var game=UnityEngine.Object.FindAnyObjectByType<GameManager>();
var score=UnityEngine.Object.FindAnyObjectByType<ScoreManager>();
var text=panel.Find("ScoreValue").GetComponent<TMPro.TMP_Text>();
if(text.text!=score.Score.ToString("N0") || !rules.IsInitialized)throw new Exception("Score ou VPE inactifs");
int colliders=model.GetComponentsInChildren<Collider>(true).Length;
var visible=model.GetComponentsInChildren<Renderer>(true).Where(r=>r.enabled).ToArray();
if(colliders!=0 || visible.Length!=3)throw new Exception("Fronton : visibilité ou colliders inattendus");
var corners=new Vector3[4];panel.GetWorldCorners(corners);var camera=Camera.main;
var points=corners.Select(camera.WorldToViewportPoint).ToArray();
System.IO.Directory.CreateDirectory("Docs/IndustriesValidation/neutral-cabinet");
ScreenCapture.CaptureScreenshot("Docs/IndustriesValidation/neutral-cabinet/Industries.png");
var report=$"scene={scene.name} paused={UnityEditor.EditorApplication.isPaused} timeScale={Time.timeScale} VPE={rules.IsInitialized}\nscore={score.Score} affichage={text.text} billes={game.BallsRemaining}\ncabinetColliders={colliders} visible={string.Join(",",visible.Select(r=>r.name))}\ncamera={camera.transform.position} rotation={camera.transform.eulerAngles} ortho={camera.orthographicSize}\nscorePanelX={points.Min(p=>p.x)}..{points.Max(p=>p.x)} scorePanelY={points.Min(p=>p.y)}..{points.Max(p=>p.y)}\n";
foreach(var renderer in visible)report+=renderer.name+" mesh="+UnityEditor.AssetDatabase.GetAssetPath(renderer.GetComponent<MeshFilter>().sharedMesh)+" materials="+string.Join(",",renderer.sharedMaterials.Select(m=>UnityEditor.AssetDatabase.GetAssetPath(m)))+"\n";
System.IO.File.WriteAllText("Docs/IndustriesValidation/neutral-cabinet/runtime.txt",report);
return report;
