if(UnityEditor.EditorApplication.isPlaying)throw new Exception("Inspecter hors Play");
var current=UnityEditor.SceneManagement.EditorSceneManager.GetActiveScene();if(current.isDirty)throw new Exception("Scène active non enregistrée");
var source=UnityEditor.SceneManagement.EditorSceneManager.OpenScene("Assets/Scenes/Neutral.unity",UnityEditor.SceneManagement.OpenSceneMode.Additive);
var report=new System.Text.StringBuilder();
foreach(var scene in new[]{source,current}) {
 var roots=scene.GetRootGameObjects();report.AppendLine("SCENE "+scene.name);
 foreach(var root in roots) {
  report.AppendLine("ROOT "+root.name+" position="+root.transform.position+" rotation="+root.transform.eulerAngles+" scale="+root.transform.lossyScale);
  foreach(var c in root.GetComponentsInChildren<Camera>(true)) {
   report.AppendLine("CAMERA "+c.name+" pos="+c.transform.position+" rot="+c.transform.eulerAngles+" ortho="+c.orthographicSize+" rect="+c.rect+" aspect="+c.aspect+" near="+c.nearClipPlane+" far="+c.farClipPlane);
   foreach(var component in c.GetComponents<Component>())report.AppendLine("  "+component.GetType().Name);
   var viewport=c.GetComponent<CabinetViewport>();if(viewport!=null)report.AppendLine(UnityEditor.EditorJsonUtility.ToJson(viewport,true));
  }
  foreach(var canvas in root.GetComponentsInChildren<Canvas>(true)) {
   report.AppendLine("CANVAS "+canvas.name+" parent="+canvas.transform.parent?.name+" mode="+canvas.renderMode+" cam="+canvas.worldCamera?.name+" sorting="+canvas.sortingOrder+" active="+canvas.gameObject.activeSelf);
   foreach(var component in canvas.GetComponents<Component>())report.AppendLine("  "+component.GetType().Name);
   var scaler=canvas.GetComponent<UnityEngine.UI.CanvasScaler>();if(scaler!=null)report.AppendLine(UnityEditor.EditorJsonUtility.ToJson(scaler,true));
   foreach(var tr in canvas.GetComponentsInChildren<RectTransform>(true))report.AppendLine("UI "+tr.name+" parent="+tr.parent?.name+" active="+tr.gameObject.activeSelf+" pos="+tr.localPosition+" rot="+tr.localEulerAngles+" size="+tr.sizeDelta+" anchor="+tr.anchorMin+"/"+tr.anchorMax+" components="+string.Join(",",tr.GetComponents<Component>().Select(x=>x==null?"MISSING":x.GetType().Name)));
   foreach(var text in canvas.GetComponentsInChildren<TMPro.TMP_Text>(true))report.AppendLine("TEXT "+text.name+" value="+text.text+" font="+text.font?.name+" material="+text.fontSharedMaterial?.name+" size="+text.fontSize+" color="+text.color);
  }
  foreach(var hud in root.GetComponentsInChildren<HudController>(true))report.AppendLine("HUD "+hud.name+" "+UnityEditor.EditorJsonUtility.ToJson(hud,true));
 }
}
System.IO.Directory.CreateDirectory("Tools/unity/out");System.IO.File.WriteAllText("Tools/unity/out/shared-presentation-before.txt",report.ToString());
UnityEditor.SceneManagement.EditorSceneManager.CloseScene(source,true);
return report.ToString();
