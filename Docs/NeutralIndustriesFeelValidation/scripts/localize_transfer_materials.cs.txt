void CopyTextures(Material material){
 foreach(var prop in material.GetTexturePropertyNames()){
  var texture=material.GetTexture(prop);if(texture==null)continue;var path=UnityEditor.AssetDatabase.GetAssetPath(texture);if(!path.StartsWith("Assets/"))continue;
  if(!copied.TryGetValue(texture,out var copy)){
   string dest=folder+"/"+System.IO.Path.GetFileName(path);if(System.IO.File.Exists(dest))throw new Exception("Copie déjà présente : "+dest);
   if(!UnityEditor.AssetDatabase.CopyAsset(path,dest))throw new Exception("Échec copie texture : "+path);
   copy=UnityEditor.AssetDatabase.LoadAssetAtPath<Texture>(dest);copied.Add(texture,copy);report.AppendLine(dest+" <- "+path);
  }
  material.SetTexture(prop,copy);
 }
}
