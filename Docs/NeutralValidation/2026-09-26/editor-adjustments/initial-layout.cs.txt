/// </summary>

    static string SceneManagerPath()=>EditorSceneManager.GetActiveScene().path;
    static void BuildRamp(string name,bool left) {
        float entry=left?-1.8f:1.05f, exit=left?-3.08f:2.36f;
        float top=left?14.9f:14.05f;
        var p=new List<Vector3>();
        p.AddRange(Curve(new Vector3(entry,0,9.5f),new Vector3(entry,0,11),new Vector3(entry,1.05f,top-1.4f),new Vector3(entry,1.05f,top)));
        p.AddRange(Curve(new Vector3(entry,1.05f,top),new Vector3(entry,1.05f,top+1.15f),new Vector3(exit,1.05f,top+1.15f),new Vector3(exit,1.05f,top)).Skip(1));
        p.AddRange(Curve(new Vector3(exit,1.05f,top),new Vector3(exit,1.05f,top-2),new Vector3(exit,1.0f,8.5f),new Vector3(exit,.72f,7.4f)).Skip(1));
        p.AddRange(Curve(new Vector3(exit,.72f,7.4f),new Vector3(exit,.42f,6.4f),new Vector3(exit,0,6.0f),new Vector3(exit,0,5.1f)).Skip(1));
        Trough("Ramp_"+name,p,left?teal:amber);
        Gate("Ramp_"+name+"_In",new Vector3(entry,.28f,9.7f),Vector3.forward,.85f);
        Gate("Ramp_"+name+"_Out",new Vector3(exit,1.30f,top-.8f),Vector3.back,.65f);
        Label("Shot_"+name,name.ToUpperInvariant(),new Vector3(entry,.025f,9.05f),.85f,left?teal.color:amber.color);
        // Actual support feet, kept outside the .64-unit ball path and above/below the playfield appropriately.
        foreach(float z in new[]{8.5f,11f,13f})Post("RampSupport_"+name+"_"+z,new Vector3(exit+(left?-.43f:.43f),0,z),.055f,.95f);
    }
    static void Gate(string name,Vector3 p,Vector3 forward,float width){var tr=gp.Find(name);if(tr==null)return;Move(tr,p);Undo.RecordObject(tr,UndoName);tr.localRotation=Quaternion.LookRotation(forward);SetBox(tr,new Vector3(width,.5f,.12f),Vector3.zero,true);}
    static void Move(Transform tr,Vector3 p){if(tr==null)return;Undo.RecordObject(tr,UndoName);tr.position=table.TransformPoint(p);}
    static void SetBox(Transform tr,Vector3 size,Vector3 center,bool trigger){var c=tr.GetComponent<BoxCollider>();if(c==null)return;Undo.RecordObject(c,UndoName);c.size=size;c.center=center;c.isTrigger=trigger;c.enabled=true;}
    static void Disable(Transform tr){if(tr==null)return;foreach(var c in tr.GetComponentsInChildren<Collider>(true)){Undo.RecordObject(c,UndoName);c.enabled=false;}foreach(var r in tr.GetComponentsInChildren<Renderer>(true)){Undo.RecordObject(r,UndoName);r.enabled=false;}}
    static Transform New(string name,Transform parent){var go=new GameObject(name);Undo.RegisterCreatedObjectUndo(go,UndoName);go.transform.SetParent(parent,false);return go.transform;}
    static Material Mat(string name,Color color,float metallic){var p=Folder+"/"+name+".mat";var m=AssetDatabase.LoadAssetAtPath<Material>(p);if(m!=null)return m;m=new Material(Shader.Find("Universal Render Pipeline/Lit")){name=name,color=color};m.SetFloat("_Metallic",metallic);m.SetFloat("_Smoothness",.38f);AssetDatabase.CreateAsset(m,p);return m;}
    static GameObject Box(string name,Vector3 p,Vector3 size,Material m,bool collision){var go=GameObject.CreatePrimitive(PrimitiveType.Cube);Undo.RegisterCreatedObjectUndo(go,UndoName);go.name=name;go.transform.SetParent(root,false);go.transform.localPosition=p;go.transform.localScale=size;go.GetComponent<Renderer>().sharedMaterial=m;if(!collision)Undo.DestroyObjectImmediate(go.GetComponent<Collider>());else go.GetComponent<Collider>().sharedMaterial=surface;return go;}
    static void Post(string name,Vector3 p,float radius,float h){var go=GameObject.CreatePrimitive(PrimitiveType.Cylinder);Undo.RegisterCreatedObjectUndo(go,UndoName);go.name=name;go.transform.SetParent(root,false);go.transform.localPosition=p+Vector3.up*h*.5f;go.transform.localScale=new Vector3(radius*2,h*.5f,radius*2);go.GetComponent<Renderer>().sharedMaterial=steel;go.GetComponent<Collider>().sharedMaterial=surface;}
    static List<Vector3> Curve(Vector3 a,Vector3 b,Vector3 c,Vector3 d){var p=new List<Vector3>();for(int i=0;i<=48;i++){float t=i/48f,u=1-t;p.Add(u*u*u*a+3*u*u*t*b+3*u*t*t*c+t*t*t*d);}return p;}
    static void Wall(string name,IEnumerable<Vector3> points,float width,float height){Sweep(name,points.ToList(),new[]{new Vector2(-width/2,0),new Vector2(width/2,0),new Vector2(width/2,height),new Vector2(-width/2,height)},steel,true);}
    static void Trough(string name,List<Vector3> p,Material accent){
        Sweep(name,p,new[]{new Vector2(-.37f,-.035f),new Vector2(.37f,-.035f),new Vector2(.37f,.52f),new Vector2(.32f,.52f),new Vector2(.32f,0),new Vector2(-.32f,0),new Vector2(-.32f,.52f),new Vector2(-.37f,.52f)},steel,true);
        foreach(int s in new[]{-1,1}){var edge=p.Select((v,i)=>v+Normal(p,i)*s*.347f+Vector3.up*.525f).ToList();Sweep(name+"_Rim"+s,edge,new[]{new Vector2(-.022f,-.018f),new Vector2(.022f,-.018f),new Vector2(.022f,.018f),new Vector2(-.022f,.018f)},accent,false);}
    }
    static Vector3 Normal(List<Vector3> p,int i){var d=p[Math.Min(p.Count-1,i+1)]-p[Math.Max(0,i-1)];return new Vector3(d.z,0,-d.x).normalized;}
    static void Sweep(string name,List<Vector3> p,Vector2[] section,Material m,bool collision){
        var v=new List<Vector3>();var tri=new List<int>();int n=section.Length;
        for(int i=0;i<p.Count;i++)foreach(var q in section)v.Add(p[i]+Normal(p,i)*q.x+Vector3.up*q.y);
        for(int i=0;i<p.Count-1;i++)for(int j=0;j<n;j++){int a=i*n+j,b=i*n+(j+1)%n;tri.AddRange(new[]{a,b,b+n,a,b+n,a+n});}
        // End caps of a U profile must follow its concavity; use three rectangles, never a solid end wall.
        int[] caps=n==8?new[]{0,1,4,0,4,5,1,2,3,1,3,4,0,5,6,0,6,7}:new[]{0,1,2,0,2,3};
        for(int i=0;i<caps.Length;i+=3){tri.AddRange(new[]{caps[i+2],caps[i+1],caps[i]});int end=(p.Count-1)*n;tri.AddRange(new[]{end+caps[i],end+caps[i+1],end+caps[i+2]});}
        var mesh=new Mesh{name=name};mesh.SetVertices(v);mesh.SetTriangles(tri,0);mesh.RecalculateNormals();mesh.RecalculateBounds();
        AssetDatabase.CreateAsset(mesh,Folder+"/"+name+".asset");var tr=New(name,root);tr.gameObject.AddComponent<MeshFilter>().sharedMesh=mesh;tr.gameObject.AddComponent<MeshRenderer>().sharedMaterial=m;
        if(collision){var c=tr.gameObject.AddComponent<MeshCollider>();c.sharedMesh=mesh;c.sharedMaterial=surface;}
    }
    static void Label(string name,string value,Vector3 p,float size,Color color){var tr=New(name,root);tr.localPosition=p;tr.localRotation=Quaternion.Euler(90,0,0);var text=Undo.AddComponent<TextMeshPro>(tr.gameObject);text.font=AssetDatabase.LoadAssetAtPath<TMP_FontAsset>("Assets/TextMesh Pro/Resources/Fonts & Materials/LiberationSans SDF.asset");text.text=value;text.fontSize=size;text.color=color;text.alignment=TextAlignmentOptions.Center;text.textWrappingMode=TextWrappingModes.NoWrap;text.rectTransform.sizeDelta=new Vector2(6,1);}
}
