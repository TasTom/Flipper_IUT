Vector2[] Boundary(float height) {
 var points=positions.Where(v=>Mathf.Abs(v.y-height)<.00001f).Select(v=>new Vector2(v.x,v.z)).Distinct().OrderBy(v=>v.x).ThenBy(v=>v.y).ToArray();
 var low=new List<Vector2>();var high=new List<Vector2>();
 foreach(var p in points){while(low.Count>1 && Cross(low[low.Count-2],low[low.Count-1],p)<=.000001f)low.RemoveAt(low.Count-1);low.Add(p);}
 foreach(var p in points.Reverse()){while(high.Count>1 && Cross(high[high.Count-2],high[high.Count-1],p)<=.000001f)high.RemoveAt(high.Count-1);high.Add(p);}
 return low.Take(low.Count-1).Concat(high.Take(high.Count-1)).ToArray();
}
float Cross(Vector2 a,Vector2 b,Vector2 c){return(b.x-a.x)*(c.y-a.y)-(b.y-a.y)*(c.x-a.x);}
void Cap(Vector2[] boundary,float height,int sign) {
 int centreIndex=vertices.Count;var centre=boundary.Aggregate(Vector2.zero,(v,p)=>v+p)/boundary.Length;
 var normal=filter.transform.localToWorldMatrix.transpose.MultiplyVector(table.TransformDirection(Vector3.up*sign)).normalized;
 AddVertex(new Vector3(centre.x,height,centre.y),normal);foreach(var p in boundary)AddVertex(new Vector3(p.x,height,p.y),normal);
 for(int i=0;i<boundary.Length;i++) {
  int a=centreIndex,b=centreIndex+1+i,c=centreIndex+1+(i+1)%boundary.Length;
  var worldNormal=Vector3.Cross(table.TransformPoint(new Vector3(boundary[i].x,height,boundary[i].y))-table.TransformPoint(new Vector3(centre.x,height,centre.y)),table.TransformPoint(new Vector3(boundary[(i+1)%boundary.Length].x,height,boundary[(i+1)%boundary.Length].y))-table.TransformPoint(new Vector3(centre.x,height,centre.y)));
  if(Vector3.Dot(worldNormal,table.TransformDirection(Vector3.up))*sign<0){int swap=b;b=c;c=swap;}
  triangles.AddRange(new[]{a,b,c});
 }
}
void AddVertex(Vector3 tablePoint,Vector3 normal) {

 int axis=Mathf.Abs(normal.y)>=Mathf.Abs(normal.x) && Mathf.Abs(normal.y)>=Mathf.Abs(normal.z)?1:Mathf.Abs(normal.x)>=Mathf.Abs(normal.z)?0:2;
 var coords=axis==0?new Vector2(local.z,local.y):axis==1?new Vector2(local.x,local.z):new Vector2(local.x,local.y);if(normal[axis]<0)coords.x=-coords.x;uv.Add(coords);
}
