from pathlib import Path
import subprocess
import re
from collections import Counter
root=Path(__file__).resolve().parents[2]
before=subprocess.run(['rtk','proxy','git','show','HEAD:Assets/Scenes/Neutral.unity'],cwd=root,capture_output=True,text=True,encoding='utf-8',check=True).stdout
after=(root/'Assets/Scenes/Neutral.unity').read_text(encoding='utf-8')
def blocks(text):
    return {(int(typ),id_):body for typ,id_,body in re.findall(r'^--- !u!(\d+) &([^\n]+)\n(.*?)(?=^--- !u!|\Z)',text,re.M|re.S)}
a,b=blocks(before),blocks(after)
changed=Counter(k[0] for k,v in a.items() if k in b and b[k]!=v)
new=Counter(k[0] for k in b if k not in a)
physical={50,54,56,64,65,135,136,143,153,154}
bad=[k for k,v in a.items() if k[0] in physical and b.get(k)!=v]
result=f'Changed existing component types: {dict(changed)}\nNew component types: {dict(new)}\nChanged/removed physical components: {bad}\n'
(root/'Tools/unity/out/vpe-scene-diff-review.txt').write_text(result,encoding='utf-8')
print(result)
if bad:raise SystemExit(1)
