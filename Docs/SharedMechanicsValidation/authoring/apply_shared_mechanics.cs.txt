 Transform CopyVisual(Transform original,string name,Transform parent,Transform origin,float factor) {
  var root=New(name,parent);
  foreach(var mf in original.GetComponentsInChildren<MeshFilter>(true)) {
   if(mf.sharedMesh==null || mf.sharedMesh.vertexCount==0)continue;
   var matrix=Matrix4x4.Scale(Vector3.one*factor)*origin.worldToLocalMatrix*mf.transform.localToWorldMatrix;
   var mesh=UnityEngine.Object.Instantiate(mf.sharedMesh);mesh.name=name+"_"+mf.name;
   mesh.vertices=mesh.vertices.Select(matrix.MultiplyPoint3x4).ToArray();
   var normals=mesh.normals;if(normals.Length>0){var normalMatrix=matrix.inverse.transpose;mesh.normals=normals.Select(n=>normalMatrix.MultiplyVector(n).normalized).ToArray();}
   // Les meshes de repos deviennent des visuels PhysX ; les générateurs/animations VPE ne sont pas copiés.
   mesh.ClearBlendShapes();mesh.RecalculateBounds();mesh.RecalculateTangents();
   UnityEditor.AssetDatabase.CreateAsset(mesh,folder+"/"+mesh.name+".asset");
   var piece=New(mf.transform==original?"Visual":mf.name,root);
   piece.gameObject.AddComponent<MeshFilter>().sharedMesh=mesh;
   var renderer=piece.gameObject.AddComponent<MeshRenderer>();var sourceRenderer=mf.GetComponent<Renderer>();renderer.sharedMaterials=sourceRenderer.sharedMaterials;
   renderer.shadowCastingMode=sourceRenderer.shadowCastingMode;renderer.receiveShadows=sourceRenderer.receiveShadows;
  }
  return root;
 }

 Bounds CombinedBounds(Transform[] roots,Transform space) {
  Bounds result=new Bounds();bool has=false;
  foreach(var root in roots)foreach(var mf in root.GetComponentsInChildren<MeshFilter>(true)) {
   if(mf.sharedMesh==null || mf.sharedMesh.vertexCount==0)continue;
   var b=TransformBounds(mf.sharedMesh.bounds,space.worldToLocalMatrix*mf.transform.localToWorldMatrix);
   if(!has){result=b;has=true;}else result.Encapsulate(b);
  }
  if(!has)throw new Exception("Géométrie absente");return result;
 }
 Bounds TransformBounds(Bounds b,Matrix4x4 matrix) {
  var result=new Bounds(matrix.MultiplyPoint3x4(b.center),Vector3.zero);
  for(int i=0;i<8;i++)result.Encapsulate(matrix.MultiplyPoint3x4(b.center+Vector3.Scale(b.extents,new Vector3((i&1)==0?-1:1,(i&2)==0?-1:1,(i&4)==0?-1:1))));
  return result;
 }
