def blocks(text):
    result = {}
    for block in re.split(r"(?=^--- !u!)", text, flags=re.M)[1:]:
        match = re.match(r"--- !u!(\d+) &(-?\d+).*\n", block)
        result[match[2]] = (match[1], block[match.end():].strip())
    return result

neutral_path = root / "Assets/Scenes/Neutral.unity"
baseline = (root / "Tools/unity/out/neutral-before-shared-mechanics.unity").read_text(encoding="utf-8")
current = neutral_path.read_text(encoding="utf-8")
old = blocks(baseline)
new = blocks(current)
changed = [key for key in old.keys() & new.keys() if old[key] != new[key]]
removed = old.keys() - new.keys()
added = new.keys() - old.keys()
protected = {"114", "54", "56", "64", "65", "135", "136"}
defaults = {"2054311436": "  tableGeometry: {fileID: 0}", "1818851149": "  compactMultiplierLabel: 0", "796815582": "  externalBallLifecycle: 0"}
def without_new_defaults(key, body):
    field = defaults.get(key)
    if field and field not in old[key][1].splitlines():
        return "\n".join(line for line in body.splitlines() if line != field)
    return body
neutral_unchanged = not removed and not any(old[key][0] in protected and old[key][1] != without_new_defaults(key, new[key][1]) for key in changed)
assert neutral_unchanged, "Unexpected existing Neutral physics/script change"

mesh_old = collections.defaultdict(list)

    if kind == "43": mesh_old[body].append(key)

    if kind == "43": mesh_new[body].append(key)
assert {body: len(ids) for body, ids in mesh_old.items()} == {body: len(ids) for body, ids in mesh_new.items()}, "Native geometry changed"
def normalize_refs(body, documents):
    def replace(match):
        key = match[1]
        item = documents.get(key)
        if item and item[0] == "43":
            return "fileID: mesh-content-" + hashlib.sha256(item[1].encode("utf-8")).hexdigest()
        return match[0]
    return re.sub(r"fileID: (-?\d+)", replace, body)

native_changed = []
lamp_instances = {"1494186559", "1830714388", "942733302", "1250472031", "1723421771"}
lamp_position_pattern = r"(target: \{fileID: 8610125003353710624, guid: 0b5e22facf585114ca5130d9e38dbdeb, type: 3\}\n      propertyPath: m_LocalPosition\.([xyz])\n      value: )([^\n]+)"
for key in industry_old.keys() & industry_new.keys():
    kind, body = industry_old[key]
    if kind not in protected | {"1001"}: continue
    now = normalize_refs(industry_new[key][1], industry_new)
    body = normalize_refs(body, industry_old)
    if key in lamp_instances:
        before_values = {m[2]: float(m[3]) for m in re.finditer(lamp_position_pattern, body)}
        after_values = {m[2]: float(m[3]) for m in re.finditer(lamp_position_pattern, now)}
        assert abs(before_values["x"] - after_values["x"]) < 0.0000002
        assert abs(before_values["z"] - after_values["z"]) < 0.0000002
        assert abs(before_values["y"] / 15.6055355 - after_values["y"]) < 0.00000001
        body = re.sub(lamp_position_pattern, lambda m: m[1] + "<native-lamp-position>", body)
        now = re.sub(lamp_position_pattern, lambda m: m[1] + "<native-lamp-position>", now)
    # Native generated mesh IDs change at save, but their contents are equal.
    if now != body:
        if kind == "114" and re.sub(r"_instanceID: \d+", "_instanceID: 0", now) == re.sub(r"_instanceID: \d+", "_instanceID: 0", body): continue
        native_changed.append((key, kind))
assert not native_changed, f"Unexpected native parameters or prefab overrides: {native_changed}"
