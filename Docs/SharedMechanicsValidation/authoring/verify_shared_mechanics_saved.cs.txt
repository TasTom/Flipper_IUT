void Check(string name,bool pass,string detail){if(!pass)failures++;report.AppendLine((pass?"PASS ":"FAIL ")+name+(detail.Length>0?" : "+detail:""));}
Bounds BoundsFor(Transform[] roots,Transform space) {
 Bounds result=new Bounds();bool has=false;
 foreach(var root in roots)foreach(var mesh in root.GetComponentsInChildren<MeshFilter>(true)) {
  if(mesh.sharedMesh==null || mesh.sharedMesh.vertexCount==0)continue;
  var b=mesh.sharedMesh.bounds;var matrix=(space!=null?space.worldToLocalMatrix:Matrix4x4.identity)*mesh.transform.localToWorldMatrix;
  for(int i=0;i<8;i++){var point=matrix.MultiplyPoint3x4(b.center+Vector3.Scale(b.extents,new Vector3((i&1)==0?-1:1,(i&2)==0?-1:1,(i&4)==0?-1:1)));if(!has){result=new Bounds(point,Vector3.zero);has=true;}else result.Encapsulate(point);}
 }
 return result;
}
